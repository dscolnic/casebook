# mission 1
- name: Atom
  aliases: [atom]   # TODO: the plurals and spellings the prose uses

# mission 1
- name: Molecule
  aliases: [molecule]   # TODO: the plurals and spellings the prose uses

# mission 1
- name: Ion

# mission 1

# mission 1
- name: Molar mass

  aliases: [kinetic energy]   # TODO: the plurals and spellings the prose uses

# mission 3
- name: Mixture

# mission 3
- name: Composition

# mission 3

# mission 3

  aliases: [mole fraction]   # TODO: the plurals and spellings the prose uses
  def: "the part of all gas particles belonging to one gas. A value of 0.68 means 68 out of every 100 gas particles are that gas."
# mission 4
- name: Valence electron

# mission 4
- name: Lewis structure
  aliases: [lewis structure]   # TODO: the plurals and spellings the prose uses

# mission 4
- name: Lone pair
  aliases: [lone pair]   # TODO: the plurals and spellings the prose uses
  def: "a pair of valence electrons not shared in a bond. Lone pairs still repel other electron groups and can change molecular shape."
# mission 4
- name: Molecular geometry
  aliases: [molecular geometry]   # TODO: the plurals and spellings the prose uses
  def: "the three-dimensional arrangement of atoms in a molecule. The shape determines whether bond effects reinforce or cancel."
# mission 4
- name: VSEPR model
  aliases: [vsepr model]   # TODO: the plurals and spellings the prose uses
  def: "predicts molecular shape by placing groups of valence electrons as far apart as possible. Lone pairs and bonds both count as electron groups."
# mission 4
- name: Polarity
  aliases: [polarity]   # TODO: the plurals and spellings the prose uses
  def: "an uneven distribution of electrical charge. A molecule is polar when its bond effects do not cancel across its shape."
# mission 4
- name: Intermolecular force
  aliases: [intermolecular force]   # TODO: the plurals and spellings the prose uses

# mission 4
- name: London dispersion force
  aliases: [london dispersion force]   # TODO: the plurals and spellings the prose uses
  def: "an attraction caused by brief shifts in electron location. Every atom and molecule has it, and particles with more electrons usually have a stronger one."
# mission 4
- name: Hydrogen bonding
  aliases: [hydrogen bonding]   # TODO: the plurals and spellings the prose uses
  def: "a strong attraction involving hydrogen bonded to nitrogen, oxygen, or fluorine and a nearby particle. It is an attraction between particles, not a new bond inside one molecule."
# mission 4
- name: Boiling
  aliases: [boiling]   # TODO: the plurals and spellings the prose uses

# mission 4
- name: Boiling point
  aliases: [boiling point]   # TODO: the plurals and spellings the prose uses

# mission 4
- name: Condensation
  aliases: [condensation]   # TODO: the plurals and spellings the prose uses

# mission 5
- name: Solution

# mission 5

# mission 5

# mission 5

# mission 5

# mission 7

# mission 7

# mission 9
